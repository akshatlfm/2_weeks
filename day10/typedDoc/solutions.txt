        from sentence_transformers import SentenceTransformer

            "My name is Akshat",
            "Akshat love Football",
            "Akshat works at QuinType."
        ]

        client = chromadb.Client()

        collection = client.create_collection(name="mydb")

        collection.add(
            ids=["e1", "e2", "e3"],
            documents=documents,
            embeddings=embeddings
        )

        query = "Where does Akshat works?"

        query_embedding = model.encode(query).tolist()

        results = collection.query(
            query_embeddings,
            n_results=2
        )

q7.
        from langchain_openai import ChatOpenAI
        from langchain_core.prompts import ChatPromptTemplate
        from langchain_core.output_parsers import StrOutputParser

        prompt = ChatPromptTemplate.from_template(
            "Write a one-sentence for this: {topic}"
        )

        model = ChatOpenAI(model="gpt-5.0-nano")

        parser = StrOutputParser()

        chain = prompt | model | parser

        response = chain.invoke({"topic": "electric scooters"})
        print(response)

q12.
        from typing import TypedDict
        from langgraph.graph import StateGraph, START, END

        class GraphState(TypedDict):
            messages: list

        def greet(state: GraphState):
            return {
                "messages": state["messages"] + ["Hello!"]
            }

        def farewell(state: GraphState):
            return {
                "messages": state["messages"] + ["Goodbye!"]
            }

        graph = StateGraph(GraphState)

        graph.add_node("greet", greet)
        graph.add_node("farewell", farewell)
        graph.add_edge(START, "greet")
        graph.add_edge("greet", "farewell")
        graph.add_edge("farewell", END)

        compiled_graph = graph.compile()

        result = compiled_graph.invoke({
            "messages": []
        })

        print(result)

        from mcp.server.fastmcp import FastMCP

        mcp = FastMCP("WordCountServer")

        @mcp.tool()
        def word_count(text: str) -> int:
            """
            Count the number of words in the given text.
            """
            return len(text.split())

        if __name__ == "__main__":
            mcp.run()
